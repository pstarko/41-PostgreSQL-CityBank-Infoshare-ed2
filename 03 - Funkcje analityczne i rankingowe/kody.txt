select 
	department_id,
	sum(salary)
from employees
where 
	department_id in (10,20,30)
group by 
	rollup(department_id)

	
select 
	department_id,
	sum(salary)
from employees
where 
	department_id in (10,20,30)
group by 
	cube(department_id)
	
	
select 
	department_id,
	job_id,
	sum(salary) sum
from employees
where 
	department_id in (10,20,30)
group by 
	rollup(department_id, job_id)
	
	
select 
	department_id,
	job_id,
	sum(salary) sum
from employees
where 
	department_id in (10,20,30)
group by 
	cube(department_id, job_id)
	
	
select 
	department_id,
	job_id,
	sum(salary) sum
from employees
where 
	department_id in (10,20,30)
	or department_id is null
group by 
	cube(department_id, job_id)
	
select 
	department_id,
	job_id,
	sum(salary) sum,
	grouping(department_id) grouping_dept,
	grouping(job_id) grouping_jobs
from employees
where 
	department_id in (10,20,30)
group by 
	cube(department_id,job_id)

select 
	department_id,
	job_id,
	sum(salary) sum
from employees
where 
	department_id in (10,20,30)
group by 
	grouping sets((department_id), (department_id, job_id), (job_id))

--Rankingi

SELECT
	last_name,
	salary,
	row_number() over (order by salary) row_number
FROM
	employees


SELECT
	last_name,
	department_id,
	salary,
	row_number() over (partition by department_id order by salary) row_number
FROM
	employees	
	
	
SELECT
	last_name,
	salary,
	rank() over (order by salary desc) rank,
	dense_rank() over (order by salary desc) drank
FROM
	employees

select
	t1.*
	from (
			SELECT
				last_name,
				salary,
				job_id,
				rank() over (partition by job_id order by salary desc) rank
				from employees) t1
where
	t1.rank = 1
