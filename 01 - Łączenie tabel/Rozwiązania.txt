/*
1.	Pokaż imię, nazwisko, numer działu i stanowisko dla wszystkich pracowników. 
Nie używaj konstrukcji „where department_id is null”, rozwiąż problem połączeniami tabel (106 wierszy)
*/

select
	*	
	from employees e join jobs j on e.job_id = j.job_id 
					 join departments d on e.department_id = d.department_id 
	
/*
2.	Pokaż pracowników pracujących w Seattle i Londynu (19 wierszy)
*/
explain
select
	e.first_name ,
	e.last_name ,
	l.city 
	from employees e join departments d on e.department_id = d.department_id 
					 join locations l on d.location_id = l.location_id 
where
	l.city in ('Seattle', 'London')
	--((city)::text = ANY ('{Seattle,London}'::text[]))

select
	e.first_name ,
	e.last_name ,
	l.city 
	from employees e join departments d on e.department_id = d.department_id 
					 join locations l on d.location_id = l.location_id and l.city in ('Seattle', 'London')


/*
3. Pokaż poniższy raport nazwisk i nazw departamentów. Uwzględnij departamenty,
które nie zatrudniają pracowników. Nie bierz pod uwagę zespołu Operations(121
wierszy)
*/

select
	e.last_name ,
	d.department_name 
	from employees e right join departments d on e.department_id = d.department_id
where
	d.department_name not in ('Operations')


/*
4. Pokaż ilość zatrudnionych pracowników w Stretford, Bombay, Sydney, London,
Munich i Seattle(6 wierszy)
*/

select
	l.city ,
	count(e.employee_id ) ilosc
	from employees e join departments d on e.department_id = d.department_id 
					 right join locations l on d.location_id = l.location_id 				 
where
	l.city in ('Stretford', 'Bombay', 'Sydney', 'London', 'Munich', 'Seattle')
group by
	l.city			


/*
5. Pokaż liczbę pracowników, którzy pracują w prowincji stanowej(State Province),
zaczynającej się od litery T (1 wiersz) (1 wiersz)
*/

explain
select
	count(e.employee_id ) ilosc
	from employees e join departments d on e.department_id = d.department_id 
					 join locations l on d.location_id = l.location_id 				 
where
	left(l.state_province, 1) = 'T'
	--substring(l.state_province , 1, 1) = 'T'
	--l.state_province like 'T%'
	--((state_province)::text ~~ 'T%'::text)


/*
6. Pokaż poniższe informacje o pracownikach, którzy są zatrudnieni na stanowisku
Accountant oraz Purchasing Clerk, których roczny dochód przekracza 37200. Roczny
dochód jest dwunastokrotnością wynagrodzenia podstawowego wraz z prowizją (5
wierszy)
*/

select
	e.last_name,
	j.job_title 
	from employees e join jobs j on e.job_id = j.job_id  
where
	j.job_title in ('Accountant', 'Purchasing Clerk')
	and (salary + (salary * coalesce(e.commission_pct ,0))) * 12 > 37200
	
/*
	7. Pokaż nazwy departamentów wraz z nazwiskami ich szefów i miejscem pracy
(11 wierszy)	
*/
	
select
	e.first_name ,
	e.last_name ,
	d.department_name ,
	l.city 
	from employees e join departments d on e.employee_id = d.manager_id 
					 join locations l on d.location_id = l.location_id 
	
					 
					 
/*
8. Zmodyfikuj powyższe zapytanie w taki sposób, by zostały wyświetlone departamenty
nie posiadające szefa. Wybierz właściwy typ połączenia tabel(27 wierszy)
*/

select
	e.first_name ,
	e.last_name ,
	d.department_name ,
	l.city 
	from employees e right join departments d on e.employee_id = d.manager_id 
					 join locations l on d.location_id = l.location_id 
	
/*
11. Pokaż ile w poszczególnych krajach wydaje się pieniędzy na wypłaty(25 wierszy)
*/
					 
select
    c.country_name ,
    coalesce(sum(salary + (salary * coalesce(commission_pct,0))),0) suma
    from employees e join departments d on  e.department_id = d.department_id
                     join locations l on l.location_id=d.location_id
					 right join countries c on c.country_id = l.country_id 
group by 
	c.country_name
order by suma desc
