SELECT
	last_name,
	hire_date,
	LAG(last_name,1,'----') OVER (ORDER BY hire_date) AS poprzednik
FROM 
	employees
	
	
SELECT
	last_name,
	hire_date,
	LAG(last_name) OVER (ORDER BY hire_date) AS poprzednik
FROM 
	employees

SELECT
	last_name,
	hire_date,
	LAG(last_name) OVER (ORDER BY hire_date, last_name ) AS poprzednik
FROM 
	employees
	
	
SELECT
	last_name,
	hire_date,
	LEAD(last_name, 2, '----') OVER (ORDER BY hire_date) AS nastepny
FROM employees
	
	
SELECT
	last_name,
	hire_date,
	job_id,
	LEAD(last_name, 1, '----') OVER (partition by job_id ORDER BY hire_date) AS nastepny
FROM employees

SELECT
	department_id,
	STRING_AGG(last_name,' ,') nazwiska
from employees
group by department_id

SELECT
	department_id,
	STRING_AGG(last_name, ',' order by last_name) nazwiska
from employees
group by department_id

select
	hire_date,
	salary,
	sum(salary) over (order by hire_date	
					  ROWS BETWEEN 1 PRECEDING 
					  		   AND 1 FOLLOWING) grow
from employees

	hire_date,
	salary,
	sum(salary) over (order by hire_date	
					  ROWS BETWEEN 2 PRECEDING 
					  		   AND current row) grow
from employees
