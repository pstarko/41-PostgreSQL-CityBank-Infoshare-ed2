/*
1. Pokaż jak kształtowała się liczba zatrudnien w poszczególnych latach. W dodatkowej
kolumnie dodaj narastające wartości ilości zatrudnień w sumie dla każdego roku (8 wierszy)
*/

select
	date_part('year', hire_date),
	count(employee_id),
	sum(count(employee_id)) over (order by date_part('year', hire_date)) narast
	from employees e 
group by
	date_part('year', hire_date)
/*
2. Zmień powyższy raport o informacje na jaki etat zostali zatrudnieni pracownicy. Również
dodaj kolumnę z wartościami narastającymi(47 wierszy)
*/
	
select
	date_part('year', hire_date),
	j.job_title,
	count(employee_id),
	sum(count(employee_id)) over (order by date_part('year', hire_date), j.job_title) narast
	from employees e join jobs j on e.job_id = j.job_id 
group by
	date_part('year', hire_date),
	j.job_title

/*
3. Posortuj dane pracowników po job_title. Pokaż first and last name oraz salary. W dodatkowej
kolumnie pokaż ile w sumie wydajemy dla całego etatu. Kolejna kolumna to procent udziału
poszczególnego pracownika w wydatkach na etat (107 wierszy)
*/

select
	t1.*,
	round(t1.salary / t1.suma_jobs * 100,2) || '%' prct
	from (
		select
			j.job_title,
			e.first_name ,
			e.last_name ,
			e.salary ,
			sum(e.salary) over (partition by j.job_title) suma_jobs
			from employees e join jobs j on e.job_id = j.job_id ) t1


/*
4. Pokaż first_name, last_name, hire_date oraz nazwisko zatrudnionego wcześniej pracownika
biorąc pod uwagę datę zatrudnienia (107 wierszy)
*/


select
	e.first_name ,
	e.last_name ,
	e.hire_date ,
	LAG(first_name || ' '|| last_name, 1, '-----') over (order by hire_date) pre
	from employees e

/*
5. Pokaż raport pracowników zatrudnionych w poszczególnych departamentach (11 wierszy)
*/


select
	d.department_name ,
	STRING_AGG(e.last_name, ' ,' order by e.last_name ) lista	
	from employees e join departments d on e.department_id = d.department_id
group by
	d.department_name 


/*
6. Zaktualizuj powyższe zapytanie w taki sposób, by otrzymać tylko pierwsze dwa nazwiska (11
wierszy)
*/


	
select
	t1.department_name ,
	string_agg(t1.last_name , ', ')
	from (
		select
			department_name,
			last_name,
			rank() over (partition by department_name order by last_name) rnk
			from employees e join departments d on e.department_id = d.department_id) t1
where
	t1.rnk <=2
group by
	t1.department_name
