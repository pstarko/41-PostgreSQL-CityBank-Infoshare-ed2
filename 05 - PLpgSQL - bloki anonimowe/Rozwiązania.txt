/*
1. Użytkownik podaje dwie liczby X i Y. Następnie pokazujesz mu liczby całkowite od X do Y.

*/

do $$
declare
	x int:=0;
	y int:=10;
begin
	LOOP		
		if x > y then
			exit;
		end if;
		RAISE NOTICE ' % ', x;
		x := x + 1;
	end loop;
end
$$;

/*
2. Stwórz pętlę pokazującą informację: 
Liczba 50, Liczba 49…., Liczba 1. 
Pętla rozpoczyna się od 50 a kończy na 1. 
Pomiń liczby parzyste pomiędzy 20 a 30

*/

DO $$
DECLARE
    i INTEGER;
BEGIN
    FOR i IN REVERSE 50..1 LOOP
    		if not (i between 20 and 30 and i % 2 = 0) then
            RAISE NOTICE 'Liczba %', i;
        end if;
    END LOOP;
END $$;


/*
Napisz skrypt, który:
• Dla każdego klienta zaktualizuje kolumnę Status w zależności od wartości Saldo:
	o VIP jeśli Saldo > 10000
	o Standard jeśli Saldo BETWEEN 1000 AND 10000
	o Zagrożony jeśli Saldo < 1000
	o Zmień dane w tabeli i wypisz je w konsoli

*/

 Id INT PRIMARY KEY,

 Nazwisko VARCHAR(50),
 Saldo DECIMAL(10, 2),

/*
Napisz skrypt, który:
• Dla każdego klienta zaktualizuje kolumnę Status w zależności od wartości Saldo:
	o VIP jeśli Saldo > 10000
	o Standard jeśli Saldo BETWEEN 1000 AND 10000
	o Zagrożony jeśli Saldo < 1000
	o Zmień dane w tabeli i wypisz je w konsoli
	Użyj: CASE

*/

DO $$
DECLARE
    klient_rec RECORD;
BEGIN
    update Klienci 
    set status = CASE
    		when saldo > 10000 then 'VIP'
    		when saldo between 1000 and 10000 then 'Standard'
    		else 'Zagrożony'
    		END;
    		
    	for 	klient_rec in select nazwisko, saldo, status from klienci
    	loop
    		raise notice 'Klient: %, Saldo: %, Status: %', 
    						klient_rec.nazwisko, klient_rec.saldo, klient_rec.status;
    	end loop;  	
END $$;

/*
5. Napisz program do obliczania kolejnych liczb całkowitych. Np. po podaniu liczby 5 otrzymasz
15(czyli 1+2+3+4+5)

*/


DO $$
DECLARE
	do_ilu int:=7;
	suma int:=0;
BEGIN
	for i in 1..do_ilu loop
		suma:= suma + i;
	end loop;
	raise notice 'Suma: %', suma;
END $$;

/*
6. Napisz program, który zwróci Ci liczbę liter i cyfr w podanych ciągu znaków i cyfr. Znajdź
wcześniej odpowiednią funkcję, która Ci to umożliwi. Do sprawdzenia czy znak jest liczbą czy
literą wykorzystaj strukturę: IF znak ~ ’[0-9]’ THEN …
*/


DO $$
DECLARE
    tekst TEXT := 'Agent007xx1'; -- Podaj tutaj swój ciąg znaków
    liczba_liter INT := 0;
    liczba_cyfr INT := 0;
    znak CHAR;
BEGIN
    -- Iteracja przez każdy znak w ciągu
    FOR i IN 1..LENGTH(tekst) LOOP
        znak := SUBSTRING(tekst,i,1); 
	raise notice '%', SUBSTRING(tekst,i,1);
	 -- Sprawdzenie czy znak jest cyfrą
       IF znak ~ '[0-9]' THEN
            liczba_cyfr := liczba_cyfr + 1;
       else
       		liczba_liter := liczba_liter + 1;
       END IF;
    END LOOP;

    -- Wyświetlenie wyników
    RAISE NOTICE 'Ciąg znaków: %', tekst;
    RAISE NOTICE 'Liczba liter: %', liczba_liter;
    RAISE NOTICE 'Liczba cyfr: %', liczba_cyfr;
END $$;
