--nr pesel (11 cyfr)
select regexp_like('12345678901xx', '^[[:digit:]]{11}$')

--b. Kod w formacie (2litery, 3 liczby, 1 litera) - tylko wielkie litery np., AG234D
select regexp_like('AG234D', '^[[:alpha:]]{2}[[:digit:]]{3}[[:alpha:]]{1}$')


/*
 *  c. Kod w formacie 2 cyfry i znak % na końcunp., - 54%
	d. Adres e-mail w formacie (imię.nazwa@domena.com)
 */


-- Kod w formacie 2 cyfry i znak % na końcu np., - 54%
SELECT REGEXP_LIKE('23%', '^[0-9]{2}%$');



-- Adres e-mail w formacie (imię.nazwa@domena.com)
SELECT REGEXP_LIKE('jan.kowalski@domena.com', '^[a-z]+\.[a-z]+@[a-z]{2,}\.com$');
/*
	2. Pokaż nazwiska pracowników (wielkość liter nie ma znaczenia):
	a. które kończą się na „i” i nie zaczynają się na „d”
 */


SELECT 
	last_name
FROM employees
WHERE 
	REGEXP_LIKE(last_name, 'i$', 'i') 
  	AND NOT REGEXP_LIKE(last_name, '^d', 'i');
	
--2b
--conajmniej 2 litery a w nazwisku
SELECT 
	last_name
FROM employees
where
	REGEXP_LIKE(last_name, '([aA].*){2,}')
