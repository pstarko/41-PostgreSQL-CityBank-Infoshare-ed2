/*
1. Wyświetl nazwiska i stanowiska pracowników pracujących w tym samym dziale co
pracownik o nazwisku Miller (5 wierszy)
*/

select
	e.first_name ,
	e.last_name ,
	(select job_title from jobs j where job_id = e.job_id ) jt
	from employees e 
where
	e.department_id = (select department_id from employees
						where last_name = 'Miller')

/*
2. Wyświetl następujące dane o najdłużej zatrudnionym pracowniku na stanowisku
FI_ACCOUNT (1 wiersz)

	*
	from employees
where
	hire_date = (select min(hire_date) from employees

/*
3. Zobacz najdłużej pracujących pracowników każdego działu. (12 wierszy)
*/

select
	e.first_name ,
	e.last_name ,
	e.hire_date ,
	e.department_id ,
	(select department_name from departments d where e.department_id = d.department_id )
	from employees e 
where
	(e.department_id , e.hire_date) in
							(select department_id, min(hire_Date)
								from employees
								group by department_id )			
					
				
/*
4. Wyświetlaj te nazwy stanowisk, w których nie zatrudniono nikogo z działu
MARKETING. Wykorzystaj podzapytanie w podzapytaniu (17 wierszy)

*/


select
	job_title
	from jobs
where
	job_id in (select job_id from employees e 
				where department_id not in (select department_id from departments
												where department_name = 'Marketing'))



/*
5. Wyświetl informacje o pracownikach zarabiających więcej niż średnia pensja dla ich
etatu(job_id) (43 wiersze)


*/

explain
select 
	e.first_name, 
	e.last_name, 
	e.salary,
	j.job_title ,
	(select avg(salary) from employees where job_id = e.job_id) avg_salary
from employees e join jobs j on e.job_id = j.job_id 
where
	salary > (select avg(salary) from employees	
				where job_id = e.job_id )


/*
7. Użyj skorelowanego podzapytania, aby wyświetlić informacje o pracowniku
niezatrudnionym w żadnym dziale (użyj właściwej formy operatora EXISTS) (1 wiersz)
*/

select 	
	*
	from employees e 
where
	not exists (select 1 from employees where department_id = e.department_id )
